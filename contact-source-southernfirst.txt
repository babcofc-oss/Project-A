        function adjustBannerPadding() {
            // Read the navigation and announcement panes directly rather than the section's
            // outerHeight - the section's own height depends on a marginTop/transform reservation
            // applied asynchronously by stickyNavBar (evergreen.app.js), so reading it here races
            // that reservation and can undercount the announcement pane. RA-2614.
            var navigationPane = $tudio('.evergreen-pane.evergreen-pane--navigation');
            var announcementPane = $tudio('.evergreen-pane.evergreen-pane--announcement');
            // Base class is always present alongside the --small-banner/--full-banner coverage
            // modifier (EvergreenLayout.cs), so target it directly instead of one modifier only.
            var layoutHeader = $tudio('.evergreen-layout-header');

            if (layoutHeader.length) {
                var headerHeight = (navigationPane.outerHeight(true) || 0) + (announcementPane.outerHeight(true) || 0);
                layoutHeader.css('--primaryNavigationModuleHeight', headerHeight + 'px');
            }
        }

        var resizeDebounceTimeout;
        $tudio(window).on("resize", function () {
            clearTimeout(resizeDebounceTimeout);
            resizeDebounceTimeout = setTimeout(adjustBannerPadding, 50);
        });

        // Recompute whenever either pane's box actually changes size - covers nav wrapping to a
        // second line and asynchronously-loaded announcement content, neither of which reliably
        // fires a window resize event. The observer is parked on the panes (same pattern as
        // stickyNavBar's announcementResizeObserver in evergreen.app.js) so a repeat run of this
        // script block - e.g. the module getting re-injected without a full page reload - replaces
        // the observer instead of stacking a second one on the same nodes. RA-2614.
        if (typeof ResizeObserver !== 'undefined') {
            var $bannerPaddingObserverTargets = $tudio('.evergreen-pane.evergreen-pane--navigation, .evergreen-pane.evergreen-pane--announcement');
            var previousBannerPaddingResizeObserver = $bannerPaddingObserverTargets.data('bannerPaddingResizeObserver');
            if (previousBannerPaddingResizeObserver) {
                previousBannerPaddingResizeObserver.disconnect();
            }

            var bannerPaddingResizeObserver = new ResizeObserver(adjustBannerPadding);
            $bannerPaddingObserverTargets.each(function () {
                bannerPaddingResizeObserver.observe(this);
            });
            $bannerPaddingObserverTargets.data('bannerPaddingResizeObserver', bannerPaddingResizeObserver);
        }

        var scrollendPollToken = 0;
        $tudio(window).on('scrollend', function () {
            var navigationPane = $tudio('.evergreen-pane.evergreen-pane--navigation');
            var announcementPane = $tudio('.evergreen-pane.evergreen-pane--announcement');
            if (!navigationPane.length) return;

            // Each scrollend supersedes any poll loop still running from a prior one, so
            // rapid/overlapping scrollend events don't leave multiple concurrent loops
            // racing to call adjustBannerPadding() off stale height snapshots. RA-2614.
            var currentPollToken = ++scrollendPollToken;
            var lastHeight = -1;
            var maxAttempts = 30;

            function waitForStableHeight() {
                if (currentPollToken !== scrollendPollToken) return;

                // Poll the same panes adjustBannerPadding reads, not the section's own
                // outerHeight, which races the async marginTop/transform reservation. RA-2614.
                var currentHeight = (navigationPane.outerHeight(true) || 0) + (announcementPane.outerHeight(true) || 0);
                if (currentHeight !== lastHeight && maxAttempts > 0) {
                    lastHeight = currentHeight;
                    maxAttempts--;
                    requestAnimationFrame(waitForStableHeight);
                } else {
                    adjustBannerPadding();
                }
            }

            requestAnimationFrame(waitForStableHeight);
        });
